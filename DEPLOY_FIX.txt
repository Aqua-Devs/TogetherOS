# Download het bestand hieronder en vervang je huidige app.py

# 2. Ga naar je togetheros folder
cd C:\Users\JOUW_NAAM\Downloads\togetheros
# ^^ Pas dit aan naar waar jouw folder staat!

git status

git add app.py

git commit -m "Fix database path for Render deployment"

git push origin main
